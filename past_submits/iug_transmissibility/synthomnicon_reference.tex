\section*{Distance Metric (\S26.1)}
\footnotesize
\textbf{Diagonal weighted Euclidean:}
$$d(A,B)=\sqrt{\sum_{i=1}^{12}w_i(\xi_i^A-\xi_i^B)^2}$$
$\mathbf{w}=(1.0^{\times 9},\,w_H{=}0.8,\,1.0,\,w_\Omega{=}0.7)$

\vspace{0.5em}
\textbf{Riemannian} (\S26.2): $d_g=\sqrt{(\boldsymbol{\xi}^A-\boldsymbol{\xi}^B)^\top\Sigma^{-1}(\boldsymbol{\xi}^A-\boldsymbol{\xi}^B)}$

\textit{Directed:} $d^\to(A\to B)_j = w_j\cdot\max(0,\xi_j^A-\xi_j^B)$ (upward free)

\section*{Core Operations}
\footnotesize
\textbf{Meet $\sqcap$ (GLB)}
\begin{itemize}[leftmargin=*,nosep,label={},topsep=4pt,itemsep=2pt]
\item Categorical: $A_i=B_i\to A_i$, else \conf
\item Ordered: $\min(A_i,B_i)$
\item \spec{$\Phi_c\sqcap x=\Phi_c$}
\end{itemize}

\textbf{Join $\sqcup$ (LUB)}
\begin{itemize}[leftmargin=*,nosep,label={},topsep=4pt,itemsep=2pt]
\item Categorical: $A_i=B_i\to A_i$, else \conf
\item Ordered: $\max(A_i,B_i)$
\end{itemize}

\textbf{Tensor $\otimes$} = Join with precondition:
$$A\otimes B = \begin{cases} A\sqcup B & \text{if no cat. conflicts} \\ \text{undefined} & \text{otherwise} \end{cases}$$
\textit{Physical:} co-assembly requires categorical compatibility.

\section*{Algebraic Properties}
\footnotesize
\begin{itemize}[leftmargin=*,nosep,topsep=4pt,itemsep=2pt]
\item \textbf{Commutative}: $A\star B = B\star A$ for $\star\in\{\sqcap,\sqcup,\otimes\}$
\item \textbf{Associative}: $(A\star B)\star C = A\star(B\star C)$ (when def.)
\item \textbf{Idempotent}: $A\star A = A$
\item \textbf{$\Phi_c$ meet-absorption}: $\Phi_c\sqcap x=\Phi_c$ for all $x$
\item \textbf{Distance}: $d(A\otimes B, A)^2 \leq d(A,B)^2$
\item \textbf{Path-dep.}: $(A\otimes B)\otimes C$ may exist when $B\otimes C$ undefined
\end{itemize}

\section*{Path-Dependence Theorem}
\footnotesize
$(A\otimes B)\otimes C$ defined even when $B\otimes C$ undefined, iff $\exists$ ordered primitive $i$:
\begin{itemize}[leftmargin=*,nosep,topsep=4pt,itemsep=2pt]
\item $C_i > B_i$ (C needs more than B)
\item $A_i = C_i$ (A supplies what C needs)
\item No categorical conflict between $(A\otimes B)$ and $C$
\end{itemize}
\textit{Interpretation:} Catalysis, scope promotion, chirality templating unified as ``context $A$ enables $B\to C$''.

\section*{Structural Universality}
\footnotesize
\textbf{Synthon:} $\langle D;T;R;P;F;K;G;\Gamma;\Phi;H;S;\Omega\rangle$

\textbf{Validity:} Must satisfy axioms A1–A4.

\textbf{Thesis:} $d(\mathbf{x},\mathbf{y})=0 \Rightarrow$ constraint-identical, substrate-independent.

\textbf{Conflict Distance:} $d_c(S)=\sqrt{|\text{Conf}(S)|}$ measures encoding reliability.

\textbf{Dual Encoding:}
\begin{itemize}[leftmargin=*,nosep,topsep=4pt,itemsep=2pt]
\item Holistic $\mathcal{E}_H(S)$: top-down from behavior
\item Compositional $\mathcal{E}_C(S)$: bottom-up from $\bigotimes S_i$
\item $\text{Conf}(S) = \{p : \mathcal{E}_H(S)[p] \neq \mathcal{E}_C(S)[p]\}$
\end{itemize}

\section*{Tensor Product Properties}
\footnotesize
When $A\otimes B$ defined:
\begin{itemize}[leftmargin=*,nosep,topsep=4pt,itemsep=2pt]
\item $(A\otimes B)_F = \max(F_A, F_B)$
\item $(A\otimes B)_K = \max(K_A, K_B)$
\item $(A\otimes B)_j = \max(A_j, B_j)$ for all ordered $j$
\item Categorical: must agree (precondition)
\item $\Phi_c$ absorption: if $A_\Phi=\Phi_c$ or $B_\Phi=\Phi_c$, then $(A\otimes B)_\Phi=\Phi_c$
\end{itemize}

\textbf{Translation Cost:} $\delta_A=d(A,A\otimes B)$, $\delta_B=d(B,A\otimes B)$; smaller $\delta$ is dominant.
