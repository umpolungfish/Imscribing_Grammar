\subsection{Imscribing Grammar --- Phonetic Imscription}

Each primitive is assigned a vocal articulation whose physics mirrors the structural meaning of the primitive. The IPA symbol(s) in brackets give the primary vocal expression.

\begin{center}
\rule{0.5\textwidth}{0.4pt}
\end{center}

\subsection{D (Dimensionality)}

\begin{itemize}
    \item \textbf{D\_wedge}: glottal stop {[}ʔ{]} --- point of complete closure, 0d articulatory event
    \item \textbf{D\_triangle}: bilabial fricative {[}ɸ{]} --- surface contact of two planes
    \item \textbf{D\_infty}: vowel continuum {[}a～ɪ～u{]} --- infinite-dimensional formant space
    \item \textbf{D\_odot}: click {[}ǃ{]} --- self-referential implosive release, inward-then-outward
\end{itemize}

\subsection{T (Topology)}

\begin{itemize}
    \item \textbf{T\_network}: trilled rhotic {[}r{]} --- branching resonances across articulation points
    \item \textbf{T\_in}: ingressive breath-hum {[}h᷅{]} --- inclusion, air drawn inward through constriction
    \item \textbf{T\_bowtie}: affricate {[}tʃ{]} --- crossing: stop$\rightarrow$fricative transition point
    \item \textbf{T\_boxtimes}: compound consonant cluster {[}k͡p{]} --- irreducible co-production
    \item \textbf{T\_odot}: ejective loop {[}kʼaʔ{]} --- self-closing articulatory circuit
\end{itemize}

\subsection{R (Relational mode)}

\begin{itemize}
    \item \textbf{R\_super}: supervisory pitch rise {[}$\nearrow$ˈ{]} --- one direction: higher structure supervenes on lower
    \item \textbf{R\_cat}: isometric vowel pair {[}ɪ$\leftrightarrow$ə{]} --- functorial mapping between positions
    \item \textbf{R\_dagger}: voiced-voiceless pair {[}bↆp{]} --- adjoint pairing within one place
    \item \textbf{R\_lr}: bidirectional sigh {[}hə$\leftrightarrow$ɑh{]} --- two-way acoustic coupling
\end{itemize}

\subsection{P (Parity/Symmetry)}

\begin{itemize}
    \item \textbf{P\_asym}: asymmetric rasp {[}ʁ̞{]} --- no symmetry preserved
    \item \textbf{P\_psi}: biphonic overtone {[}ə+ə̇{]} --- superposed fundamental+overtone
    \item \textbf{P\_pm}: voicing pair {[}z$\leftrightarrow$s{]} --- Z2 voiced/voiceless flip
    \item \textbf{P\_sym}: neutral schwa {[}ə{]} --- all symmetries preserved, maximal neutrality
    \item \textbf{P\_pm\_sym}: Frobenius snap {[}ǀt͡s{]} --- $\mu$∘$\delta$=id at the critical crossing point
\end{itemize}

\subsection{F (Fidelity)}

\begin{itemize}
    \item \textbf{F\_ell}: deterministic tap {[}ɾ{]} --- classical, no coherence needed
    \item \textbf{F\_eth}: breathy voice {[}ɦ{]} --- thermal noise mixed with phonation
    \item \textbf{F\_hbar}: coherent whistle {[}çʷ{]} --- quantum-phase-stable laminar tone
\end{itemize}

\subsection{K (Kinetics)}

\begin{itemize}
    \item \textbf{K\_fast}: driven plosive {[}k{]} --- rapid energy release
    \item \textbf{K\_mod}: moderate approximant {[}ɹ{]} --- intermediate timescale
    \item \textbf{K\_slow}: prolonged nasal hum {[}ŋ̊ː{]} --- near-equilibrium sustained tone
    \item \textbf{K\_trap}: frozen-order formant {[}aː{]} --- trapped ordered resonance
    \item \textbf{K\_MBL}: creaky voice {[}a̰{]} --- frozen disorder, irregular vibration
\end{itemize}

\subsection{G (Scope)}

\begin{itemize}
    \item \textbf{G\_beth}: local dental tap {[}ɾ̪{]} --- nearest-neighbor articulation
    \item \textbf{G\_gimel}: mesoscale pharyngeal {[}ʕ{]} --- intermediate range constriction
    \item \textbf{G\_aleph}: universal glottal {[}h{]} --- long-range, whole-tract participation
\end{itemize}

\subsection{Gamma (Interaction grammar)}

\begin{itemize}
    \item \textbf{G\_and}: simultaneous labial-velar {[}k͡p{]} --- conjunctive co-articulation
    \item \textbf{G\_or}: fluctuating rhotic {[}r～ɽ{]} --- alternate articulator paths
    \item \textbf{G\_seq}: sequential diphthong {[}ɑɪ{]} --- ordered temporal progression
    \item \textbf{G\_broad}: broadcast exclamation {[}aː{]} --- one-to-all projection
\end{itemize}

\subsection{Phi (Criticality)}

\begin{itemize}
    \item \textbf{Phi\_sub}: dead stop {[}ʔ{]} --- no criticality, no scaling
    \item \textbf{Phi\_c}: critical creak {[}ˀ{]} --- self-modeling gate, critical divergence
    \item \textbf{Phi\_c\_complex}: vibrato {[}a̪ː̃{]} --- complex-plane oscillation
    \item \textbf{Phi\_EP}: exceptional merge {[}a̸{]} --- non-Hermitian degeneracy, two modes coalesce
    \item \textbf{Phi\_super}: runaway howl {[}a͈{]} --- supercritical, no bounded attractor
\end{itemize}

\subsection{H (Chirality)}

\begin{itemize}
    \item \textbf{Ħ_Ñ}: memoryless click {[}ǀ{]} --- no prior state needed
    \item \textbf{Ħ_£}: one-step echo {[}a.ə{]} --- one temporal memory
    \item \textbf{Ħ_A}: two-step cadence {[}a.ə.ɑ̃{]} --- two prior states retained
    \item \textbf{H\_inf}: eternal drone {[}oːːː{]} --- no finite Markov boundary
\end{itemize}

\subsection{S (Stoichiometry)}

\begin{itemize}
    \item \textbf{one\_one}: unique phoneme {[}ǂ{]} --- one type, one instance
    \item \textbf{n\_n}: reduplicated mora {[}ma.ma{]} --- many identical units
    \item \textbf{n\_m}: heterogeneous syllable {[}stɹɑ{]} --- multiple distinct phoneme types
\end{itemize}

\subsection{Omega (Winding)}

\begin{itemize}
    \item \textbf{Omega\_0}: unwound utterance {[}pa{]} --- trivial topology
    \item \textbf{Omega\_Z2}: binary parity chirp {[}pↆpʰaↆa{]} --- Z2 protected, requires Ħ_A+
    \item \textbf{Omega\_Z}: winding hum {[}ã$\searrow$ã$\nearrow$ãː{]} --- integer topological winding number
    \item \textbf{Omega\_NA}: non-Abelian braid {[}ǂ͡ǀ$\leftrightarrow$ǀ͡ǂ{]} --- path-dependent braiding requires D\_odot
\end{itemize}
